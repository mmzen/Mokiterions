//! Throwaway evidence harness for WO-MOK-007. Not a shipped test; deleted after capture.
use mokiterions_tui::options::{self, Startup};
use mokiterions_tui::render::draw;
use mokiterions_tui::state::Observer;
use ratatui::Terminal;
use ratatui::backend::TestBackend;
use ratatui::buffer::Buffer;
use ratatui::style::{Color, Modifier};

fn start(args: &[&str]) -> Observer {
    match options::parse(args.to_vec()).unwrap() {
        Startup::Run(o) => Observer::new(o).unwrap(),
        Startup::Help => panic!(),
    }
}

fn frame_of(observer: &mut Observer, w: u16, h: u16) -> Buffer {
    let mut t = Terminal::new(TestBackend::new(w, h)).unwrap();
    t.draw(|f| draw(f, observer)).unwrap();
    t.backend().buffer().clone()
}

fn name(c: Color) -> &'static str {
    match c {
        Color::Green => "green ",
        Color::Indexed(208) => "orange",
        Color::Red => "red   ",
        Color::Reset => "-     ",
        _ => "?     ",
    }
}

fn dump(observer: &mut Observer, w: u16, h: u16, note: &str) {
    let buffer = frame_of(observer, w, h);
    println!("--- {w} x {h}: {note}");
    for y in buffer.area.top()..buffer.area.bottom() {
        let cells: Vec<_> = (0..46.min(w)).map(|x| buffer.cell((x, y)).unwrap()).collect();
        let text: String = cells.iter().map(|c| c.symbol()).collect();
        if text.trim().is_empty() {
            continue;
        }
        // One band letter per cell, so a band that leaked into an unstyled cell is visible.
        let bands: String = cells
            .iter()
            .map(|c| match c.fg {
                Color::Green => 'G',
                Color::Indexed(208) => 'O',
                Color::Red => 'R',
                _ => '.',
            })
            .collect();
        let rev = if cells.iter().all(|c| c.modifier.contains(Modifier::REVERSED)) {
            " REVERSED"
        } else {
            ""
        };
        if bands.contains(['G', 'O', 'R']) {
            println!("y{y:>2} |{text}|{rev}");
            println!("     |{bands}|");
        } else {
            println!("y{y:>2} |{text}|{rev}");
        }
    }
    println!();
}

        println!(

            a.id, a.health, b(a.health), a.satiety, b(a.satiety), a.energy, b(a.energy)
        );
    }
    println!();
