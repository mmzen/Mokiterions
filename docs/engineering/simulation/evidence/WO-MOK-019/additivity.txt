# ---- full text of this script, retained as VER-MOK-012 requires ----

"""VER-MOK-012 scenario 12: re-run every configuration retained under WO-MOK-002, WO-MOK-010 and
WO-MOK-011, and compare bytes.

Usage, from the repository root:

    python docs/engineering/simulation/evidence/WO-MOK-019/analysis/prior-captures.py \
        <binary> <evidence-root> <output-file>

`REQ-MOK-045`'s matrix row asks that "every run retained under `WO-MOK-002`, `WO-MOK-010` and
`WO-MOK-011` reproduces byte for byte at the configuration recorded with it". Taken literally that is
unsatisfiable, and the reason is worth stating plainly rather than working around quietly: **two of
those work orders retained pre-change captures of an older text stream.** `WO-MOK-010` added `fear`
and `waste_tolerance` to the stream and `WO-MOK-011` added `name`, each under an approved amendment.
A capture taken before those additions cannot be reproduced by a build that makes them, and no
correct engine could reproduce it.

So the retained runs are sorted into the three shapes the text stream has had, and each is compared
in the only way that can carry information:

  * a capture of the **current** shape must be byte-identical to the re-run, with nothing removed
    from either side. This is the comparison that says `WO-MOK-019` did not disturb the text stream,
    and it is the one `REQ-MOK-045` is really about;
  * a capture of an **older** shape must be byte-identical to the re-run *after the fields added
    since it are deleted*, and the deletion must be a no-op on the capture itself. That second half
    is what stops the projection from hiding a difference instead of removing an addition, and it is
    `WO-MOK-010`'s own safeguard, reused here.

A capture that fails either comparison is a reproduction failure. A capture that passes the second
but not the first is not: it is a capture of an older stream, and the report says so per file rather
than in a footnote.

## Where each generation and each projection comes from

The generations are detected from the capture's own bytes -- the `agent_initialized` line names the
attributes that existed when it was written -- and not from the directory it sits in, so a file filed
in the wrong place is classified by its content.

The projection from the current shape down to generation 2 deletes `name:<value>,` and nothing else.
`WO-MOK-011`'s `analysis/name-occurrences.txt` measured that `name:` occurs twelve times in a full
run, all on `agent_initialized` lines at tick 0, so the pattern is anchored on that event's
`result=` prefix and cannot match a value, an identifier or another field.

The projection down to generation 1 additionally deletes the three patterns
`WO-MOK-010/baseline/projection.py` defines, restated here with the same anchoring and in the same
order -- the transition form before the scalar form, because the scalar pattern is a prefix of it.

## The configuration recorded with each capture

`WO-MOK-011` encodes it in the file name. `WO-MOK-010` renamed its retained subset and dropped the
density, so the density is measured back out of the capture: `SPEC-MOK-001`'s resource density fixes
the initial endowment at `hundredths * 8192 / 10000` per territory, so counting `food_initialized`
lines identifies the density uniquely among the three the matrix declares. If the re-run then matches,
the inference is confirmed by the comparison rather than asserted.

`WO-MOK-002` retained no run stream at all. What it retained is handled at the bottom of this script,
in the only terms it can be: the one raw stream line its `manual-observation.md` quotes.
"""

import io
import os
import re
import subprocess
import sys

CELLS_PER_TERRITORY = 128 * 64
DECLARED_DENSITIES = ('0.15', '0.75', '1.50')

# The current shape's additions, newest first. Deleting a prefix of this list projects the current
# stream back to the shape that existed before those additions were made.
NAME = re.compile(r'(?<=event=agent_initialized result=)name:[A-Za-z]+,')
FEAR_TRANSITION = re.compile(r',fear:\d+->\d+')
WASTE_TOLERANCE = re.compile(r',waste_tolerance:\d+')
FEAR_SCALAR = re.compile(r',fear:\d+')


def to_generation_2(text):
    """Delete what WO-MOK-011 added: the name, and nothing else."""
    return NAME.sub('', text)


def to_generation_1(text):
    """Delete what WO-MOK-011 and WO-MOK-010 added, in WO-MOK-010's order."""
    text = to_generation_2(text)
    text = FEAR_TRANSITION.sub('', text)
    text = WASTE_TOLERANCE.sub('', text)
    return FEAR_SCALAR.sub('', text)


PROJECTIONS = {3: lambda text: text, 2: to_generation_2, 1: to_generation_1}

GENERATIONS = {
    3: 'name, fear and waste_tolerance — the current shape, from WO-MOK-011',
    2: 'fear and waste_tolerance, no name — the shape between WO-MOK-010 and WO-MOK-011',
    1: 'neither — the shape before WO-MOK-010',
}


def generation_of(text):
    """Which shape a capture was written in, from the capture's own bytes."""
    for line in text.split('\n'):
        if 'event=agent_initialized' in line:
            if 'name:' in line:
                return 3
            return 2 if 'waste_tolerance:' in line else 1
    return None


def density_of(text):
    """The density a capture was taken at, measured from its initial endowment."""
    endowment = text.count('event=food_initialized')
    for density in DECLARED_DENSITIES:
        whole, _, fraction = density.partition('.')
        hundredths = int(whole) * 100 + int(fraction)
        if 2 * (hundredths * CELLS_PER_TERRITORY // 10000) == endowment:
            return density
    return None


def configuration(path, text):
    """(seed, policy, density, ticks, trace) for one retained capture.

    `WO-MOK-011`'s names carry everything. `WO-MOK-010`'s carry the policy, the seed and the tick
    count, and the density is measured from the capture.
    """
    name = os.path.basename(path)
    stem = name.rsplit('.', 1)[0]
    parts = stem.split('-')

    if name.endswith('.log'):
        # {pre|post}-{policy}-seed{S}-ticks{N}[-trace].log
        policy = parts[1]
        seed = parts[2].removeprefix('seed')
        ticks = parts[3].removeprefix('ticks')
        trace = 'trace' in parts[4:]
        return seed, policy, density_of(text), ticks, trace

    # seed{S}-{policy}-d{D}-trace{off|on}.txt
    seed = parts[0].removeprefix('seed')
    policy = parts[1]
    density = parts[2].removeprefix('d')
    trace = parts[3] == 'traceon'
    return seed, policy, density, '1000', trace


def rerun(binary, seed, policy, density, ticks, trace, cache):
    """One run of the current build, cached by configuration."""
    key = (seed, policy, density, ticks, trace)
    if key not in cache:
        arguments = [
            binary, '--seed', seed, '--ticks', ticks, '--policy', policy, '--density', density
        ]
        if trace:
            arguments.append('--trace-actions')
        completed = subprocess.run(arguments, capture_output=True)
        if completed.returncode != 0:
            raise SystemExit(f'{key} exited {completed.returncode}: {completed.stderr!r}')
        cache[key] = completed.stdout.decode('utf-8')
    return cache[key]


def compare(capture, rerun_text, generation, whole):
    """The verdict for one capture. Returns (verdict, detail)."""
    projected = PROJECTIONS[generation](rerun_text)

    # WO-MOK-010's safeguard: the projection must not touch the capture. A projection that changed a
    # byte of it would be erasing a difference rather than an addition.
    if PROJECTIONS[generation](capture) != capture:
        return 'PROJECTION NOT A NO-OP', 'the projection altered the retained capture itself'

    if whole:
        matched = projected == capture
    else:
        matched = projected.startswith(capture)

    if matched:
        if generation == 3:
            return 'identical', ''
        return 'identical after projection', f'projected to generation {generation}'

    # A failure names the line, not the run.
    left = projected.split('\n')
    right = capture.split('\n')
    detail = f'{len(left)} re-run lines against {len(right)} retained'
    for number, (a, b) in enumerate(zip(left, right), start=1):
        if a != b:
            detail = f'first difference at line {number}: re-run {a!r} against retained {b!r}'
            break
    return 'DIFFERS', detail


def main():
    binary, evidence, output_file = sys.argv[1:4]
    binary = os.path.abspath(binary)

    # Every retained run stream, by work order. `init` captures are the initialization prefix of a
    # run and are compared as a prefix; everything else is compared whole.
    groups = [
        ('WO-MOK-010', 'baseline/full', '.log', True),
        ('WO-MOK-010', 'post/full', '.log', True),
        ('WO-MOK-011', 'baseline/init', '.txt', False),
        ('WO-MOK-011', 'baseline/full', '.txt', True),
        ('WO-MOK-011', 'post/init', '.txt', False),
        ('WO-MOK-011', 'post/full', '.txt', True),
    ]

    cache = {}
    rows = []
    for work_order, subdirectory, suffix, whole in groups:
        directory = os.path.join(evidence, work_order, subdirectory)
        for name in sorted(os.listdir(directory)):
            if not name.endswith(suffix):
                continue
            path = os.path.join(directory, name)
            capture = io.open(path, 'r', encoding='utf-8', newline='').read()
            generation = generation_of(capture)
            seed, policy, density, ticks, trace = configuration(path, capture)
            if density is None:
                rows.append(
                    (work_order, subdirectory, name, '', 0, 'DENSITY NOT IDENTIFIED',
                     f'{capture.count("event=food_initialized")} food_initialized lines match no '
                     'declared density')
                )
                continue
            text = rerun(binary, seed, policy, density, ticks, trace, cache)
            verdict, detail = compare(capture, text, generation, whole)
            rows.append(
                (
                    work_order,
                    subdirectory,
                    name,
                    f'seed={seed} policy={policy} d={density} ticks={ticks} '
                    f'trace={"on" if trace else "off"}',
                    generation,
                    verdict,
                    detail,
                )
            )

    # ---------------------------------------------------------------------------------------
    # WO-MOK-002. No run stream was retained, so there is nothing to compare whole. Its
    # `manual-observation.md` quotes one raw stream line, and that line is checkable.
    # ---------------------------------------------------------------------------------------
    quoted = (
        'tick=0 subject=F0019 event=food_initialized result=class:low,position:17:1,territory:A'
    )
    observation = os.path.join(evidence, 'WO-MOK-002', 'manual-observation.md')
    quoted_is_there = quoted in io.open(observation, 'r', encoding='utf-8').read()
    text = rerun(binary, '42', 'reference', '0.75', '20', True, cache)
    wo002 = [
        'WO-MOK-002 retained no run stream: its evidence is a cargo test log and nine documents.',
        f'the line manual-observation.md quotes is still quoted there: '
        f'{"yes" if quoted_is_there else "NO — the quotation has changed"}',
        f'the same line appears in a re-run at seed 42, density 0.75%, reference, traced: '
        f'{"yes" if quoted in text else "NO"}',
        'its 20-tick excerpt is a hand-trimmed digest -- fields were dropped and columns aligned by '
        'hand -- so it is not a capture and cannot be byte-compared to anything.',
    ]

    # ---------------------------------------------------------------------------------------
    # One configuration under both build profiles, so that "byte-identical output from any build
    # of one commit" is checked rather than relied on. Every capture script in this repository
    # says it, because it is what lets a release build capture what a debug build asserts.
    # ---------------------------------------------------------------------------------------
    other = binary.replace('release', 'debug') if 'release' in binary else binary.replace(
        'debug', 'release'
    )
    cross = []
    if os.path.exists(other) and other != binary:
        arguments = ['--seed', '42', '--ticks', '1000', '--policy', 'individual', '--density', '1.50']
        first = subprocess.run([binary, *arguments], capture_output=True)
        second = subprocess.run([other, *arguments], capture_output=True)
        cross = [
            f'{os.path.basename(os.path.dirname(binary))} against '
            f'{os.path.basename(os.path.dirname(other))}, '
            f'seed 42, 1.50%, individual, 1,000 ticks:',
            f'  exit codes {first.returncode} and {second.returncode}',
            f'  {len(first.stdout)} bytes and {len(second.stdout)} bytes',
            f'  identical: {"yes" if first.stdout == second.stdout else "NO"}',
        ]
    else:
        cross = [f'only one profile is built ({binary}); no cross-build comparison was made']

    # ---------------------------------------------------------------------------------------
    # The write-up.
    # ---------------------------------------------------------------------------------------
    failures = [row for row in rows if row[5] not in ('identical', 'identical after projection')]
    by_verdict = {}
    for row in rows:
        by_verdict.setdefault(row[5], 0)
        by_verdict[row[5]] += 1

    lines = [
        '# VER-MOK-012 scenario 12: every retained prior capture, re-run and byte-compared',
        '#',
        f'# binary: {binary}',
        f'# evidence root: {evidence}',
        '# command: python docs/engineering/simulation/evidence/WO-MOK-019/analysis/'
        'prior-captures.py <binary> <evidence-root> <output-file>',
        '#',
        f'# {len(rows)} retained captures, {len(cache)} distinct configurations re-run.',
        '#',
        '# The text stream has had three shapes. Each capture is classified by its own bytes and',
        '# compared against the re-run projected back to that shape:',
    ]
    for generation in sorted(GENERATIONS, reverse=True):
        lines.append(f'#   generation {generation}: {GENERATIONS[generation]}')
    lines.append('#')
    lines.append('# A generation-3 capture is compared with nothing removed from either side. An')
    lines.append('# older one is compared after the fields added since were deleted, and the')
    lines.append('# deletion is separately required to be a no-op on the capture itself.')
    lines.append('')

    for work_order, subdirectory, name, config, generation, verdict, detail in rows:
        row = (
            f'{work_order}/{subdirectory}/{name:<40} gen{generation} {config:<52} {verdict}'
        )
        if detail:
            row += f'  {detail}'
        lines.append(row)

    lines.append('')
    lines.append('=' * 100)
    lines.append('# verdicts')
    lines.append('=' * 100)
    for verdict in sorted(by_verdict):
        lines.append(f'  {by_verdict[verdict]:>4}  {verdict}')
    lines.append('')
    lines.append(f'# {len(rows)} captures, {len(failures)} failing')
    lines.append('')
    lines.append('=' * 100)
    lines.append('# WO-MOK-002')
    lines.append('=' * 100)
    for line in wo002:
        lines.append(f'  {line}')
    lines.append('')
    lines.append('=' * 100)
    lines.append('# one commit, two build profiles')
    lines.append('=' * 100)
    for line in cross:
        lines.append(f'  {line}')
    lines.append('')
    lines.append(f'# result: {"FAIL" if failures else "PASS"}')
    lines.append('')
    lines.append('# ---- full text of this script, retained as VER-MOK-012 requires ----')
    lines.append('')
    lines.extend(io.open(__file__, 'r', encoding='utf-8').read().split('\n'))

    with io.open(output_file, 'w', encoding='utf-8', newline='\n') as handle:
        handle.write('\n'.join(lines) + '\n')

    print(
        f'{len(rows)} captures, {len(cache)} configurations re-run, {len(failures)} failing; '
        + ', '.join(f'{count} {verdict}' for verdict, count in sorted(by_verdict.items()))
    )
    return 1 if failures else 0


if __name__ == '__main__':
    sys.exit(main())
